#!/bin/bash

# 宿主机上的源公钥文件路径
HOST_KEYS="/root/.ssh/authorized_keys"

if [ ! -f "$HOST_KEYS" ]; then
    echo "错误：宿主机 $HOST_KEYS 文件不存在！"
    exit 1
fi

echo "=== 开始增量同步 SSH 公钥 ==="

# --------------------------------------------------
# 1. 处理普通 LXC 容器（精确过滤掉 Template）
# --------------------------------------------------
echo -e "\n>>> 检查 LXC 容器..."

# 准确判断：通过 config 中的 template 标记过滤，确保只拉取非 Template 容器
LXC_IDS=""
for ctid in $(pct list | awk 'NR>1 {print $1}'); do
    # 检查配置文件是否包含 template: 1
    if ! grep -q -E "^template:\s*1" "/etc/pve/lxc/${ctid}.conf" 2>/dev/null; then
        LXC_IDS="$LXC_IDS $ctid"
    fi
done

if [ -z "$(echo $LXC_IDS | tr -d ' ')" ]; then
    echo "  └─ 未检测到可用的非模板 LXC 容器。"
else
    for id in $LXC_IDS; do
        echo "[LXC $id] 正在检查..."
        
        # 确保容器内部 .ssh 目录存在
        pct exec $id -- mkdir -p /root/.ssh
        pct exec $id -- chmod 700 /root/.ssh
        pct exec $id -- touch /root/.ssh/authorized_keys
        pct exec $id -- chmod 600 /root/.ssh/authorized_keys

        # 逐行读取宿主机的 key，对比后追加
        ADDED_COUNT=0
        while IFS= read -r key || [ -n "$key" ]; do
            # 忽略空行和注释
            [[ -z "$key" || "$key" =~ ^# ]] && continue
            
            # 检查 key 是否已存在于该 LXC 中
            if ! pct exec $id -- grep -qF "$key" /root/.ssh/authorized_keys; then
                echo "$key" | pct exec $id -- tee -a /root/.ssh/authorized_keys >/dev/null
                ((ADDED_COUNT++))
            fi
        done < "$HOST_KEYS"

        if [ $ADDED_COUNT -gt 0 ]; then
            echo "  └─ 成功增量追加了 $ADDED_COUNT 个新 Key。"
        else
            echo "  └─ 已包含所有 Key，无需修改。"
        fi
    done
fi

# --------------------------------------------------
# 2. 处理 QEMU 虚拟机（VM）
# --------------------------------------------------
echo -e "\n>>> 检查 QEMU 虚拟机 (VM)..."

VM_IDS=$(qm list 2>/dev/null | awk 'NR>1 && $3=="running" {print $1}')

if [ -z "$VM_IDS" ]; then
    echo "  └─ 当前没有正在运行的 QEMU 虚拟机，自动跳过。"
else
    for id in $VM_IDS; do
        # 检查是否开启了 qemu-guest-agent
        if qm guest exec $id -- true &>/dev/null; then
            echo "[VM $id] 正在检查 (via Guest Agent)..."
            
            qm guest exec $id -- mkdir -p /root/.ssh
            qm guest exec $id -- chmod 700 /root/.ssh
            qm guest exec $id -- touch /root/.ssh/authorized_keys
            qm guest exec $id -- chmod 600 /root/.ssh/authorized_keys

            ADDED_COUNT=0
            while IFS= read -r key || [ -n "$key" ]; do
                [[ -z "$key" || "$key" =~ ^# ]] && continue
                
                if ! qm guest exec $id -- grep -qF "$key" /root/.ssh/authorized_keys &>/dev/null; then
                    qm guest exec $id -- bash -c "echo '$key' >> /root/.ssh/authorized_keys"
                    ((ADDED_COUNT++))
                fi
            done < "$HOST_KEYS"

            if [ $ADDED_COUNT -gt 0 ]; then
                echo "  └─ 成功增量追加了 $ADDED_COUNT 个新 Key。"
            else
                echo "  └─ 已包含所有 Key，无需修改。"
            fi
        else
            echo "[VM $id] 未运行 QEMU Guest Agent，已跳过。"
        fi
    done
fi

# --------------------------------------------------
# 3. 专门处理模板容器 (Template)
# --------------------------------------------------
echo -e "\n>>> 专门处理模板容器 (Template)..."
TEMPLATES="500" # 需要写入的模板 ID

for tpl_id in $TEMPLATES; do
    CONF_FILE="/etc/pve/lxc/${tpl_id}.conf"
    if [ -f "$CONF_FILE" ]; then
        echo "[Template $tpl_id] 静默注入公钥..."
        
        # 判断 RootFS 挂载路径（支持标准 local/local-lvm 等）
        ROOTFS="/var/lib/lxc/$tpl_id/rootfs"
        
        if [ -d "$ROOTFS/root" ]; then
            mkdir -p "$ROOTFS/root/.ssh"
            chmod 700 "$ROOTFS/root/.ssh"
            touch "$ROOTFS/root/.ssh/authorized_keys"
            
            ADDED_COUNT=0
            while IFS= read -r key || [ -n "$key" ]; do
                [[ -z "$key" || "$key" =~ ^# ]] && continue
                if ! grep -qF "$key" "$ROOTFS/root/.ssh/authorized_keys"; then
                    echo "$key" >> "$ROOTFS/root/.ssh/authorized_keys"
                    ((ADDED_COUNT++))
                fi
            done < "$HOST_KEYS"
            chmod 600 "$ROOTFS/root/.ssh/authorized_keys"
            
            if [ $ADDED_COUNT -gt 0 ]; then
                echo "  └─ 模板 $tpl_id 成功同步了 $ADDED_COUNT 个新 Key。"
            else
                echo "  └─ 模板已包含所有 Key，无需修改。"
            fi
        else
            echo "  └─ 提示：模板 $tpl_id 的 rootfs 为 LVM/ZFS 块设备，离线未挂载，建议开机后运行或从其克隆即可。"
        fi
    fi
done

echo -e "\n=== 同步完成！ ==="
