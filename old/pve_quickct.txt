#!/bin/bash

TEMPLATE_ID="500"

echo "=== PVE LXC 快速克隆一键脚本 ==="

# 1. 获取 CTID
read -p "请输入新容器的 CTID (例如 601): " NEW_ID
if [ -z "$NEW_ID" ]; then
    echo "错误：CTID 不能为空！"
    exit 1
fi

if pct status "$NEW_ID" &>/dev/null; then
    echo "错误：CTID $NEW_ID 已经被占用！"
    exit 1
fi

# 2. 获取容器名称
read -p "请输入新容器的名称 (例如 LXC-Homepage): " CT_NAME
if [ -z "$CT_NAME" ]; then
    echo "错误：容器名称不能为空！"
    exit 1
fi

# 3. 克隆容器
echo -e "\n[1/3] 正在从模板 $TEMPLATE_ID 克隆到 $NEW_ID ($CT_NAME)..."
pct clone "$TEMPLATE_ID" "$NEW_ID" --hostname "$CT_NAME" --full 1
if [ $? -ne 0 ]; then
    echo "错误：克隆失败！"
    exit 1
fi

# 4. 首次启动（触发 Hook 脚本自动写入 TUN 配置）
echo "[2/3] 正在启动容器 $NEW_ID..."
pct start "$NEW_ID"

# 5. 等待 2 秒由宿主机直接执行重新加载
echo "[3/3] 等待 2 秒，由宿主机强制执行重启以生效 TUN..."
sleep 2
pct reboot "$NEW_ID"

echo -e "\n=== 容器 $NEW_ID ($CT_NAME) 创建并重启完毕！ ==="
